
# 1. Import Required Libraries
import numpy as np
import random

# 2. Define Tic-Tac-Toe Environment
class TicTacToe:

    def __init__(self):
        self.reset()

    # Initialize the board
    def reset(self):
        self.board = [0] * 9
        return tuple(self.board)

    # Display the board
    def display(self):
        symbols = {0: ' ', 1: 'X', -1: 'O'}

        for i in range(0, 9, 3):
            print(" | ".join(symbols[x] for x in self.board[i:i+3]))
            if i < 6:
                print("---------")

    # Get available positions
    def available_moves(self):
        return [i for i in range(9) if self.board[i] == 0]

    # Check winner
    def check_winner(self):
        win_positions = [
            (0, 1, 2), (3, 4, 5),
            (6, 7, 8), (0, 3, 6),
            (1, 4, 7), (2, 5, 8),
            (0, 4, 8), (2, 4, 6)
        ]

        for a, b, c in win_positions:
            if self.board[a] == self.board[b] == self.board[c] != 0:
                return self.board[a]

        if 0 not in self.board:
            return 0

        return None

    # Perform a move
    def step(self, action, player):
        self.board[action] = player
        return self.check_winner()


# 3. Build Reinforcement Learning Model
class QLearningAgent:

    def __init__(self, alpha=0.1, gamma=0.9,
                 epsilon=1.0, epsilon_decay=0.9995,
                 epsilon_min=0.01):

        self.q_table = {}
        self.alpha = alpha
        self.gamma = gamma
        self.epsilon = epsilon
        self.epsilon_decay = epsilon_decay
        self.epsilon_min = epsilon_min

    # Get Q-value
    def get_q(self, state, action):
        return self.q_table.get((state, action), 0.0)

    # Select action
    def choose_action(self, state, available_moves):
        if random.random() < self.epsilon:
            return random.choice(available_moves)

        q_values = [self.get_q(state, a) for a in available_moves]
        max_q = max(q_values)

        best_actions = [
            a for a, q in zip(available_moves, q_values)
            if q == max_q
        ]

        return random.choice(best_actions)

    # Update Q-table
    def learn(self, state, action, reward, next_state,
              next_moves, done):

        old_q = self.get_q(state, action)

        if done:
            future_q = 0
        else:
            future_q = max(
                [self.get_q(next_state, a) for a in next_moves],
                default=0
            )

        new_q = old_q + self.alpha * (
            reward + self.gamma * future_q - old_q
        )

        self.q_table[(state, action)] = new_q

    # Reduce exploration
    def decay_epsilon(self):
        self.epsilon = max(
            self.epsilon_min,
            self.epsilon * self.epsilon_decay
        )


# 4. Train the Model
env = TicTacToe()
agent = QLearningAgent()

episodes = 50000

for episode in range(episodes):

    state = env.reset()
    done = False

    while not done:

        # AI plays as X
        action = agent.choose_action(
            state, env.available_moves()
        )

        result = env.step(action, 1)
        next_state = tuple(env.board)

        if result is not None:
            reward = 1 if result == 1 else 0
            agent.learn(
                state, action, reward,
                next_state, [], True
            )
            break

        # Random opponent plays as O
        opponent_action = random.choice(
            env.available_moves()
        )

        result = env.step(opponent_action, -1)
        next_state = tuple(env.board)

        if result is not None:
            reward = -1 if result == -1 else 0
            agent.learn(
                state, action, reward,
                next_state, [], True
            )
            break

        # Continue learning
        next_moves = env.available_moves()

        agent.learn(
            state, action, 0,
            next_state, next_moves, False
        )

        state = next_state

    agent.decay_epsilon()

print("Training Completed Successfully!")
print("Total Episodes:", episodes)
print("Total Q-Table States:", len(agent.q_table))


# 5. Test the Trained Model
agent.epsilon = 0  # Disable random exploration

while True:

    env.reset()

    print("\nNew Game Started!")
    env.display()

    while True:

        # AI move
        state = tuple(env.board)
        available = env.available_moves()

        action = agent.choose_action(state, available)
        result = env.step(action, 1)

        print("\nAI Move:")
        env.display()

        if result is not None:
            if result == 1:
                print("AI Wins!")
            else:
                print("Match Draw!")
            break

        # Human move
        available = env.available_moves()

        print("\nAvailable Positions:",
              [x + 1 for x in available])

        try:
            human_action = int(
                input("Enter your position (1-9): ")
            ) - 1

            if human_action not in available:
                print("Invalid Move! Try Again.")
                continue

        except ValueError:
            print("Please enter a number between 1 and 9.")
            continue

        result = env.step(human_action, -1)

        print("\nYour Move:")
        env.display()

        if result is not None:
            if result == -1:
                print("You Win!")
            else:
                print("Match Draw!")
            break

    again = input("\nPlay Again? (yes/no): ").lower()

    if again != "yes":
        print("Thank you for playing!")
        break
